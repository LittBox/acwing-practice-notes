# 字符串基础题（AcWing 5288 / 3640 / 5054）

## 1) 翻转字符串（AcWing 5288）
- 链接： https://www.acwing.com/problem/content/5291/

### ✅ 代码（C++）
```c++
#include <iostream>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;
    for (int i = (int)s.length() - 1; i >= 0; i--) {
        cout << s[i];
    }
    return 0;
}
```

## 2) 删除子串（AcWing 3640）
- 链接： https://www.acwing.com/problem/content/3643/

### ✅ 代码（C++）
```c++
#include <iostream>
#include <string>
using namespace std;

int main() {
    string s;
    cin >> s;
    string out;
    int n = s.length();
    for (int i = 0; i < n; ) {
        if (i + 2 < n && s[i] == 'g' && s[i+1] == 'z' && s[i+2] == 'u') {
            i += 3;
            continue;
        }
        out += s[i++];
    }
    cout << out;
    return 0;
}
```

## 3) 拼接字符串（AcWing 5054）
- 链接： https://www.acwing.com/problem/content/5057/

### ✅ 代码（C++）
```c++
#include <iostream>
#include <string>
#include <set>
#include <vector>
using namespace std;

string normalize(const string& s) {
    string res;
    for (char c : s) {
        if (c == '-' || c == '_' || c == ';') continue;
        if (c >= 'A' && c <= 'Z') c = c - 'A' + 'a';
        res += c;
    }
    return res;
}

int main() {
    vector<string> ss(3);
    for (int i = 0; i < 3; i++) cin >> ss[i];
    int n;
    cin >> n;

    set<string> patterns;
    vector<string> norm(3);
    for (int i = 0; i < 3; i++) norm[i] = normalize(ss[i]);
    patterns.insert(norm[0] + norm[1] + norm[2]);
    patterns.insert(norm[0] + norm[2] + norm[1]);
    patterns.insert(norm[1] + norm[0] + norm[2]);
    patterns.insert(norm[1] + norm[2] + norm[0]);
    patterns.insert(norm[2] + norm[1] + norm[0]);
    patterns.insert(norm[2] + norm[0] + norm[1]);

    while (n--) {
        string s;
        cin >> s;
        string t = normalize(s);
        cout << (patterns.count(t) ? "ACC" : "WA") << endl;
    }
    return 0;
}
```
