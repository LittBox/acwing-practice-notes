# [791. 高精度加法](https://www.acwing.com/problem/content/793/) 

## ❌ 问题 1：`A.size() || B.size()`

`||` 是 逻辑或，结果只会是` 0 `或` 1`。

例如：
```c++
A.size() = 3
B.size() = 5

A.size() || B.size() = 1
```
所以循环最多只执行 1 次。

正确应该是取 最大长度：
```c++
max(A.size(), B.size())
```

## ❌ 问题 2：数组越界

这里：
```c++
t = A[i] + B[i] + t;
```
如果 `i >= A.size() 或 i >= B.size()`就会 越界访问。

必须判断。
<br>✅ 正确写法<br>
```c++
vector<int> add(vector<int> &A, vector<int> &B){
    int t = 0;
    vector<int> C;

    for(int i = 0; i < max(A.size(), B.size()); i++){
        if(i < A.size()) t += A[i];
        if(i < B.size()) t += B[i];

        C.push_back(t % 10);
        t /= 10;
    }

    if(t) C.push_back(t);

    return C;
}
```

# [792. 高精度减法](https://www.acwing.com/problem/content/794/)
## AC代码 
```c++
#include<iostream>
#include<vector>

using namespace std;
//判断A >= B
bool cmp(vector<int> &A, vector<int> &B){
    if(A.size() != B.size()) return A.size() > B.size();
    for(int i = A.size() - 1; i >= 0; i--){
        if(A[i] != B[i]) return A[i] > B[i];
    }
    return true;
} 
// A >= B
vector<int> sub(vector<int> &A, vector<int> &B){
    vector<int> C;
    
    for(int i = 0,t = 0; i < A.size(); i++){
        t = A[i] - t;
        if(i < B.size()) t -= B[i];
        C.push_back((t + 10) % 10);
        if(t < 0) t = 1;
        else t = 0;
    }
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}
int main(){
    string a,b;
    vector<int> A,B;
    cin >> a >> b;
    for(int i = a.size() - 1; i >= 0; i--) A.push_back(a[i] - '0');
    for(int i = b.size() - 1; i >= 0; i--) B.push_back(b[i] - '0');
    //判断A,B的大小，这个函数自己实现
    vector<int> C;
    if(cmp(A,B)){
        // A - B > 0
        C = sub(A,B);
        for(int i = C.size() - 1; i >= 0; i--){
            printf("%d",C[i]);
        }
    }else{
        // A - B < 0, 变成B - A， 输出时带"-"
        C = sub(B,A);
        printf("%c",'-');
        for(int i = C.size() - 1; i >= 0; i--){
            printf("%d",C[i]);
        }
    }
    
    return 0;
}
```

# [793. 高精度乘法](https://www.acwing.com/problem/content/description/795/)

## ✅AC代码
```c++
#include<iostream>
#include<vector>

using namespace std;

vector<int> mul(vector<int> &A, int b){
    vector<int> C;
    int t = 0;
    for(int i = 0; i < A.size() || t;i++){   //这一块的判断条件和加法有所不同
        if(i < A.size()) t += A[i] * b;
        C.push_back(t % 10);
        t /= 10;
    }
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
}

int main(){
    string a;
    int b;
    cin >> a >> b;
    vector<int> A;
    for(int i = a.size() - 1; i >= 0 ; i--) A.push_back(a[i] - '0');
    auto C = mul(A,b);
    for(int i = C.size() - 1; i >= 0; i--){
        printf("%d",C[i]);
    }
    return 0;
}
```

# [794. 高精度除法](https://www.acwing.com/problem/content/796/)

 ## 不是“一模一样”。第二份里有一个关键错误，导致 A[i] 越界访问，从而触发 segmentation fault。

问题在这里：

第一份是：
```c++
for(int i = a.size() - 1; i >= 0; i--) A.push_back(a[i] - '0');
```
第二份你写成了：
```C++
for(int i = a.size() - 1; i >= 0; i++) A.push_back(a[i] - '0');
```
注意最后是 `i++`，不是``` i--```。

## ✅AC代码

```c++
#include<iostream>
#include<vector>
#include<algorithm>
using namespace std;

vector<int> divide(vector<int> &A, int b, int &t){
    vector<int> C;
    
    for(int i = A.size() - 1; i >= 0; i--){
        t = t * 10 + A[i];
        C.push_back(t / b);
        t %= b;
    }
    reverse(C.begin(),C.end());
    while(C.size() > 1 && C.back() == 0) C.pop_back();
    return C;
    
}

int main(){
    string a;
    int b;
    vector<int> A;
    cin >> a >> b;
    for(int i = a.size() -1; i >= 0; i--) A.push_back(a[i] - '0');
    int t = 0;
    auto C = divide(A,b,t);
    for(int i = C.size() - 1; i >= 0; i--){
        printf("%d",C[i]);
    }
    cout << endl << t <<endl;
    
    
    return 0;
}
```

```c++
#include<iostream>
#include<vector>

using namespace std;

int main(){
    int m,n,q;//m*n 二维矩阵 q个询问数
    scanf("%d%d%d",&m,&n,&q);
    // [1,1] -> [m,n] 
    vector<vector<int>> A(m+1,vector<int>(n+1));
    for(int i = 1; i <= m; i++){
        for(int j = 1; j <= n; j++){
            scanf("%d",&A[i][j]);
        }
    }
    //求前缀和数组
    vector<vector<int>> S(m+1,vector<int>(n+1));
    for(int i = 1; i <=m ; i++){
        for(int j = 1; j <= n; j++)
            S[i][j] = S[i-1][j] + S[i][j-1] - S[i-1][j-1] + A[i][j];
    }
    //求查询子矩阵和
    while(q--){
        int x1,y1,x2,y2;
        scanf("%d%d%d%d",&x1,&y1,&x2,&y2);
        cout << S[x2][y2] - S[x1-1][y2] - S[x2][y1-1] + S[x1-1][y1-1] << endl;
    }
    
}
```

# [798. 差分矩阵](https://www.acwing.com/problem/content/800/)

## ❌b 开小了，会越界

你在 insert 里有：
```c++
b[x2 + 1][y1] -= c;
b[x1][y2 + 1] -= c;
b[x2 + 1][y2 + 1] += c;
```
所以当 x2 = m 或 y2 = n 时，会访问到：
```C++
b[m + 1][...]
b[...][n + 1]
```
但你现在只开了：
```c++
vector<vector<int>> b(m+1, vector<int>(n+1));
```
下标范围只有 0 ~ m 和 0 ~ n，访问 m+1 / n+1 就越界了。

应该开成：
```C++
vector<vector<int>> b(m + 2, vector<int>(n + 2, 0));
```
a 也建议一起开大一点：
```C++
vector<vector<int>> a(m + 2, vector<int>(n + 2, 0));
```


## ❌前缀还原那里写错了

你写的是：
```c++
b[i] += b[i-1];
```
但 b[i] 是一整行 vector<int>，不能这么加。

二维差分恢复原矩阵，要做二维前缀和：
```c++
b[i][j] = b[i][j] + b[i - 1][j] + b[i][j - 1] - b[i - 1][j - 1];
```
## AC代码

```c++
#include<iostream>
#include<vector>

using namespace std;
 
void insert(vector<vector<int>> &b,int x1, int y1,int x2, int y2,int c ){
    b[x1][y1] += c;
    b[x2 + 1][y1] -= c;
    b[x1][y2+1] -= c;
    b[x2+1][y2+1] +=c;
}
 
int main(){
    int m,n,q;
    scanf("%d%d%d",&m,&n,&q);
    vector<vector<int>> a(m+2,vector<int>(n+2));
    for(int i = 1; i <= m; i++)
        for(int j = 1; j <= n ; j++)
            cin >> a[i][j];
            
    vector<vector<int>> b(m+2,vector<int>(n+2));
    for(int i= 1; i <= m; i++)
        for(int j = 1; j <= n; j++)
            insert(b,i,j,i,j,a[i][j]);
            
    while(q--){
        int x1,y1,x2,y2,c;
        scanf("%d%d%d%d%d",&x1,&y1,&x2,&y2,&c);
        insert(b,x1,y1,x2,y2,c);
    }
    for(int i = 1; i <= m; i++)
        for(int j = 1; j <= n; j++)
            b[i][j] = b[i-1][j] + b[i][j-1] - b[i-1][j-1] + b[i][j] ;
    
    for(int i = 1; i <= m ; i++){
        for(int j = 1; j <= n; j++)
            cout << b[i][j] << " ";
        cout << endl;
    }
        
    return 0;
}
```
