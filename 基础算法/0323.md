[803. 区间合并](https://www.acwing.com/problem/content/805/)

```c++
#include <iostream>
#include <algorithm>

using namespace std;

typedef pair<int, int> PII;
vector<PII> segs;

void merge(vector<PII> &segs){
    int st = -2e9, ed = -2e9;
    vector<PII> res;
    sort(segs.begin(),segs.end()); //先比较first的大小,再比较second的大小
    for(auto seg : segs){
        if(ed < seg.first){
            if(ed != -2e9) res.push_back({st,ed});
            st = seg.first, ed = seg.second;
        }else{
            ed = max(ed, seg.second);
        }
    }
    if(st != -2e9) res.push_back({st,ed});
    segs = res;
}

int main(){
    int n;
    scanf("%d", &n);
    for(int i = 0; i < n; i++){
        int l,r;
        cin >> l >> r;
        segs.push_back({l,r});
    }
   
    merge(segs);
    
    cout << segs.size() ;
    
    return 0;
}
```



[ 842. 排列数字](https://www.acwing.com/activity/content/problem/content/905/)

主要错误:
1. 没有判断某个数是否已经用过

你虽然用了 st[i] 记录状态，但在循环里没有写：
```c++
if (st[i]) continue;
```
所以每一层都会把 1~n 全部再选一遍，结果不是全排列，而是会生成 n^n 种序列。

2. 输出格式不太对

你现在写的是：
```c++
printf("%d", path[i]);
```
这样会连在一起输出，比如 123、132。通常题目要求每个数字之间有空格，并且每组排列换行。

3. 缺少命名空间

你用了 cin，但没有写：
```c++
using namespace std;
```
或者改成：
```c++
std::cin >> n;
```


```c++
#include <iostream>

using namespace std;

const int N = 10;

int st[N];  //记录状态
int path[N];
int n;

void dfs(int u){ //这个是层数
    if(u == n){
        for(int i = 0; i < n; i++) printf("%d ",path[i]);
        printf("\n");
        return;
    }
    for(int i = 1; i <= n; i++){
        if(!st[i]){
            path[u] = i; 
            st[i] = true;
            dfs(u + 1);
            st[i] = false;
        }
      
    }
    
}

int main(){
    
    cin >> n;
    dfs(0);
    return 0;
    
}
```
